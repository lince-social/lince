cmake_minimum_required(VERSION 3.19)
include(cmake/board.cmake)
set(PROJECT_VER "1.0.0")
option(LINCE_COMPILE_TESTS "Compile firmware test cases without executing them" OFF)
option(LINCE_RUN_TESTS "Run firmware tests on boot instead of starting workflows" OFF)
if(LINCE_RUN_TESTS)
    set(LINCE_COMPILE_TESTS ON)
endif()
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(lince_ailuros)
if(LINCE_COMPILE_TESTS AND NOT CONFIG_UNITY_ENABLE_64BIT)
    message(FATAL_ERROR "Firmware tests require CONFIG_UNITY_ENABLE_64BIT=y")
endif()
idf_build_set_property(COMPILE_OPTIONS "-Werror" APPEND)
